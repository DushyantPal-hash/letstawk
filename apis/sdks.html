GET STARTED

SDK in Action - Python Example

Here's how simple it is to initiate a voice call with the Letstawk Python SDK.

# Install: pip install letstawk

from letstawk import Letstawk
from letstawk.models import CallRequest

# Initialize the client
client = Letstawk(api_key="your_api_key")

# Create a call request
call = CallRequest(
  to="+1234567890",
  from="+1234567890",
  flow_id="qualification_flow_123",
  context={
    "lead_name": "John Doe",
    "source": "website"
  }
)

# Initiate the call
response = client.calls.create(call)
print(response.call_id)
# => "call_abc123xyz"
// Install: npm install letstawk

import { Letstawk, CallRequest } from 'letstawk';

// Initialize the client
const client = new Letstawk({
  apiKey: 'your_api_key'
});

// Create a call request
const call = new CallRequest({
  to: '+1234567890',
  from: '+1234567890',
  flowId: 'qualification_flow_123',
  context: {
    leadName: 'John Doe',
    source: 'website'
  }
});

// Initiate the call
const response = await client.calls.create(call);
console.log(response.callId);
// => "call_abc123xyz"
